cmake_minimum_required(VERSION 3.24)
project(WingetSwiftModern LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(SLINT_FEATURE_BACKEND_QT OFF CACHE BOOL "" FORCE)
set(SLINT_FEATURE_INTERPRETER OFF CACHE BOOL "" FORCE)
set(SLINT_FEATURE_SYSTEM_TRAY OFF CACHE BOOL "" FORCE)
set(SLINT_FEATURE_TESTING OFF CACHE BOOL "" FORCE)
set(SLINT_FEATURE_BACKEND_WINIT ON CACHE BOOL "" FORCE)
set(SLINT_FEATURE_RENDERER_SOFTWARE ON CACHE BOOL "" FORCE)
set(SLINT_FEATURE_RENDERER_FEMTOVG OFF CACHE BOOL "" FORCE)
set(SLINT_FEATURE_RENDERER_FEMTOVG_WGPU OFF CACHE BOOL "" FORCE)
set(SLINT_FEATURE_RENDERER_SKIA OFF CACHE BOOL "" FORCE)
set(SLINT_FEATURE_RENDERER_SKIA_OPENGL OFF CACHE BOOL "" FORCE)
set(SLINT_FEATURE_RENDERER_SKIA_VULKAN OFF CACHE BOOL "" FORCE)
set(SLINT_STYLE fluent CACHE STRING "" FORCE)
set(DEFAULT_SLINT_EMBED_RESOURCES embed-files CACHE STRING "" FORCE)
include(FetchContent)
FetchContent_Declare(Slint URL https://codeload.github.com/slint-ui/slint/tar.gz/refs/tags/v1.18.0 SOURCE_SUBDIR api/cpp)
FetchContent_MakeAvailable(Slint)
add_executable(WingetSwiftModern WIN32 src/modern.cpp)
target_compile_definitions(WingetSwiftModern PRIVATE UNICODE _UNICODE SLINT_STATIC)
target_compile_options(WingetSwiftModern PRIVATE /utf-8 /O2 /EHsc)
target_link_libraries(WingetSwiftModern PRIVATE Slint::Slint comctl32 comdlg32 shell32 user32 gdi32 advapi32 dwmapi uxtheme imm32)
target_link_options(WingetSwiftModern PRIVATE /MANIFEST:EMBED "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/app.manifest")
slint_target_sources(WingetSwiftModern ui/app.slint)
set_target_properties(WingetSwiftModern PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/dist/modern")
